cmake_minimum_required(VERSION 3.15)
project(camelot_openkeyscan C)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WITH_OPENMP OFF CACHE BOOL "" FORCE)
set(WITH_LSR_BINDINGS OFF CACHE BOOL "" FORCE)
set(WITH_VR32 OFF CACHE BOOL "" FORCE)
set(WITH_AVFFT OFF CACHE BOOL "" FORCE)
set(WITH_CR32 ON CACHE BOOL "" FORCE)
set(WITH_CR64 ON CACHE BOOL "" FORCE)
set(WITH_CR32S ON CACHE BOOL "" FORCE)
set(WITH_CR64S ON CACHE BOOL "" FORCE)
set(WITH_PFFFT ON CACHE BOOL "" FORCE)
set(WITH_HI_PREC_CLOCK ON CACHE BOOL "" FORCE)
set(WITH_FLOAT_STD_PREC_CLOCK OFF CACHE BOOL "" FORCE)
set(WITH_DEV_TRACE ON CACHE BOOL "" FORCE)
set(LIB_SUFFIX "" CACHE STRING "" FORCE)
add_subdirectory(../../vendor/soxr soxr)
add_library(camelot_soxr_half STATIC half.c)
target_include_directories(camelot_soxr_half PRIVATE ../../vendor/soxr/src)
set_target_properties(camelot_soxr_half PROPERTIES C_STANDARD 99)
install(TARGETS camelot_soxr_half ARCHIVE DESTINATION lib)
